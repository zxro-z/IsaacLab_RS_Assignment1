step_count=0;max_step=0.;max_episode=0.

  for r in csv.DictReader(stream):

  assert float(e['final_displacement'])==float(old['forward_displacement_m'])
  assert np.isfinite(components).all() and np.isfinite(total).all()
  err=np.abs(components.sum(1)-total);assert np.allclose(components.sum(1),total,rtol=1e-6,atol=1e-6)
  max_step=max(max_step,float(err.max()));step_count+=n
  summed=components.sum(0);recorded=np.array([float(e[t+'_sum']) for t in m.TERMS])
  assert np.allclose(summed,recorded,atol=1e-12)
  native=float(old['episode_return']);assert np.isclose(summed.sum(),native,rtol=2e-6,atol=1e-6)
  max_episode=max(max_episode,abs(float(summed.sum())-native))
  hit=int(first[policy,i]['first_reach_step_40m']) if first[policy,i]['reach_40m']=='True' else n
  assert int(e['pre40_steps'])==hit and int(e['post40_steps'])==n-hit
  pre=components[:hit].sum(0);post=components[hit:].sum(0)
  assert np.allclose(pre,[float(e[t+'_pre40']) for t in m.TERMS],atol=1e-12)
  assert np.allclose(post,[float(e[t+'_post40']) for t in m.TERMS],atol=1e-12)
  values[i,j,0]=np.r_[summed,native]
  values[i,j,1]=np.r_[pre,total[:hit].sum()];values[i,j,2]=np.r_[post,total[hit:].sum()]
  values[i,j,3]=values[i,j,0]/n;values[i,j,4]=values[i,j,0]/(n*result['control_dt']);lengths[i,j]=n

# Independent fixed-RNG multiplicities and elementwise weighted sums reproduce all paired CIs.
rng=np.random.default_rng(2404);weights=rng.multinomial(100,np.full(100,.01),size=50000)
for ref,new in PAIRS:
 name=f'{POLICIES[ref]} → {POLICIES[new]}'
 for s,scope in enumerate(m.MEASURES):
  for k,term in enumerate(m.TERMS+('total_reward_from_env',)):
   delta=values[:,new,s,k]-values[:,ref,s,k];samples=np.sum(weights*delta[None,:],axis=1)/100
   r=result['pairwise'][name][scope][term]
   assert np.isclose(r['observed_delta'],delta.mean(),atol=1e-12)
   assert np.isclose(r['median_paired_delta'],np.median(delta),atol=1e-12)
   assert np.allclose([r['ci95_low'],r['ci95_high']],np.quantile(samples,[.025,.975]),atol=1e-12)
   assert np.isclose(r['bootstrap_median_delta'],np.median(samples),atol=1e-12)

 rate=(values[:,new,3,:7]-values[:,ref,3,:7])*(lengths[:,new,None]+lengths[:,ref,None])/2
 assert np.allclose(duration+rate,values[:,new,0,:7]-values[:,ref,0,:7],atol=1e-12)
 for scope,d in [('duration_associated',duration),('rate_associated',rate)]:

   samples=np.sum(weights*d[:,k][None,:],axis=1)/100;r=result['pairwise'][name][scope][term]
   assert np.allclose([r['ci95_low'],r['ci95_high']],np.quantile(samples,[.025,.975]),atol=1e-12)
for r in result['total_delta_reconciliation']:
